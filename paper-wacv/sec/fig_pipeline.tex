  flow/.style={->, line width=0.5pt, gray!70},
]
\def\W{8.4}       % total width (cm)
\def\bw{0.29}     % bar pitch
\def\bh{0.95}     % bar area height

%% ── the untouched training run ─────────────────────────────────────────
\node[box, fill=gray!8, text width=\W cm-6pt] (train)
  {\textbf{Standard training run} --- optimizer, LR schedule, early stopping
   and epoch budget \emph{unchanged}};

%% ── checkpoint set, as per-epoch selection scores ──────────────────────
\coordinate (o) at ($(train.south west)+(1.30,-1.42)$);
\foreach \i/\h/\sel in {%
  1/0.12/0, 2/0.28/0, 3/0.40/0, 4/0.58/0, 5/0.70/0, 6/0.84/1,
  7/0.76/0, 8/0.90/1, 9/0.80/1, 10/0.87/1, 11/0.72/0, 12/0.82/1,
  13/0.68/0, 14/0.64/0, 15/0.60/0, 16/0.58/0}{
  \ifnum\sel=1\def\c{selcol}\else\def\c{black!22}\fi
  \fill[\c] ($(o)+(\i*\bw-\bw,0)$) rectangle ++(\bw*0.70,\h*\bh);
}
\draw[black!45, line width=0.35pt] (o) -- ++(16*\bw,0);
\draw[dashed, winbg, line width=0.5pt]
  ($(o)+(11*\bw,-0.05)$) rectangle ++(5*\bw,\bh*1.04);
\node[winbg, anchor=north, font=\tiny] at ($(o)+(13.5*\bw,-0.07)$)
  {last-$k$ window};
\node[anchor=east, align=right, font=\tiny] at ($(o)+(-0.05,\bh*0.5)$)
  {validation\\score};
\node[anchor=south, selcol, font=\tiny] at ($(o)+(9*\bw,0.93*\bh)$) {top-$k$};
\node[anchor=north, font=\tiny] at ($(o)+(8*\bw,-0.34)$)
  {one checkpoint per epoch, each with its validation score};

%% ── criterion box ──────────────────────────────────────────────────────
\node[box, fill=white, draw=black!45, anchor=west, align=left, font=\tiny]
  (crit) at ($(o)+(16.4*\bw,\bh*0.52)$)
  {\textbf{rank by the score}\\\textbf{the pipeline already uses}\\[1.5pt]
   classification: validation loss\\
   detection / seg.: fitness};

%% ── the two selection rules ────────────────────────────────────────────
\node[box, fill=basecol!8, draw=basecol, anchor=north west, text width=2.9cm]
  (base) at ($(train.south west)+(0,-2.62)$)
  {\textbf{Baseline} ($k{=}1$)\\[1pt]
   keep the single best checkpoint};
\node[box, fill=selcol!8, draw=selcol, anchor=north east, text width=4.5cm]
  (ours) at ($(train.south east)+(0,-2.62)$)
  {\textbf{Ours} (top-$k$)\\[1pt]
   average the $k$ best checkpoints,\\then recompute BN statistics};

%% ── evaluation ─────────────────────────────────────────────────────────
\node[box, fill=gray!8, text width=\W cm-6pt, anchor=north]
  (eval) at ($(base.south)!0.5!(ours.south)+(0,-0.34)$)
  {\textbf{Final evaluation} --- held-out split, single-model inference cost};

%% ── arrows ─────────────────────────────────────────────────────────────
\draw[flow] (train.south) -- ++(0,-0.26);

\draw[flow] (base.south) -- (base.south |- eval.north);
\draw[flow] (ours.south) -- (ours.south |- eval.north);
\end{tikzpicture}}
\caption{The rule changes only the last step of the pipeline. A standard run is
  left untouched; its checkpoints are ranked by the validation score the
  pipeline already computes, and instead of keeping the single best we average
  the best $k$. Because the ranking is by score and not by position, the chosen
  checkpoints need not be consecutive and need not sit at the end of training
  (\cref{sec:selection}). Bar heights are schematic.}
