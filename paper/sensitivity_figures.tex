% ── Annotation macro ─────────────────────────────────────────────────────
% \Ann{anchor}{x}{y}{val}{std}{color}
\newcommand{\Ann}[6]{%
  \node[anchor=#1, text=#6,
        font=\fontsize{5.8}{7}\selectfont\bfseries,
        fill=white, fill opacity=0.88, text opacity=1,
        inner sep=1pt, rounded corners=1.2pt]
    at (axis cs:#2,#3)
    {$#4$\\[-1.5pt]\normalfont$\pm\!#5$};}
